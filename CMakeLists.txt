cmake_minimum_required(VERSION 3.24)
project(ShDisplayViewer VERSION 1.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
option(SHDISPLAY_ENABLE_EARTH "Build the FreeEarth map backend" ON)
option(SHDISPLAY_ENABLE_VIDEO "Build RTSP support" ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/private/FreeEarthSDK/CMakeLists.txt")
    set(_sdk_default SOURCE)
else()
    set(_sdk_default PACKAGE)
endif()
set(SHDISPLAY_EARTH_PROVIDER "${_sdk_default}" CACHE STRING "SOURCE (private development) or PACKAGE (public client)")
set_property(CACHE SHDISPLAY_EARTH_PROVIDER PROPERTY STRINGS SOURCE PACKAGE)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Network OpenGL OpenGLWidgets Xml)
qt_standard_project_setup(REQUIRES 6.11)
if(MSVC)
    add_compile_options(/utf-8 /W4)
    add_compile_definitions(NOMINMAX _USE_MATH_DEFINES)
endif()
if(SHDISPLAY_ENABLE_EARTH)
    if(SHDISPLAY_EARTH_PROVIDER STREQUAL "SOURCE")
        set(FREEEARTHSDK_BUILD_SHARED ON CACHE BOOL "Build private SDK binaries")
        set(FREEEARTHSDK_BUILD_LEGACY_MODULES ON CACHE BOOL "Retain plot algorithms")
        set(FREEEARTHSDK_BUILD_LEGACY_PLUGINS ON CACHE BOOL "Retain map service drivers")
        add_subdirectory(private/FreeEarthSDK)
        if(TARGET FreeEarthLegacyPluginsAll)
            add_dependencies(FreeEarthRuntime FreeEarthLegacyPluginsAll)
        endif()
    elseif(SHDISPLAY_EARTH_PROVIDER STREQUAL "PACKAGE")
        find_package(FreeEarthSDK 4.0 CONFIG REQUIRED)
    else()
        message(FATAL_ERROR "Invalid SHDISPLAY_EARTH_PROVIDER")
    endif()
endif()
if(SHDISPLAY_ENABLE_VIDEO)
    if(WIN32)
        set(VIDEOSDK_FFMPEG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/ffmpeg/windows" CACHE PATH "FFmpeg 7 headers and import libraries")
        set(VIDEOSDK_FFMPEG_RUNTIME_DIR "${VIDEOSDK_FFMPEG_ROOT}/bin" CACHE PATH "FFmpeg runtime")
    endif()
    add_subdirectory(sdk/videoSDK)
endif()
file(GLOB_RECURSE _client_sources CONFIGURE_DEPENDS
    src/app/*.cpp src/app/*.h src/ui/*.cpp src/ui/*.h
    src/modules/zqbs/*.cpp src/modules/zqbs/*.h src/modules/zqbs/*.ui
    src/modules/zqts/*.cpp src/modules/zqts/*.h)
list(FILTER _client_sources EXCLUDE REGEX "/\\._[^/]+$")
if(MSVC)
    # Localized /showIncludes output can be misdecoded by Ninja. Keep explicit
    # project header dependencies so object allocation/layout changes cannot
    # leave MainWindow or test callers compiled against stale class definitions.
    set(_client_headers ${_client_sources})
    list(FILTER _client_headers INCLUDE REGEX "\\.h$")
    set(_client_cpp ${_client_sources})
    list(FILTER _client_cpp INCLUDE REGEX "\\.cpp$")
    set_property(SOURCE ${_client_cpp} src/main.cpp APPEND
        PROPERTY OBJECT_DEPENDS ${_client_headers})
endif()
add_library(ShDisplayCore STATIC ${_client_sources})
target_include_directories(ShDisplayCore PUBLIC src)
target_link_libraries(ShDisplayCore PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::OpenGL Qt6::OpenGLWidgets Qt6::Xml)
if(SHDISPLAY_ENABLE_EARTH)
    target_compile_definitions(ShDisplayCore PUBLIC ZQXXXT_ENABLE_EARTH)
    target_link_libraries(ShDisplayCore PUBLIC FreeEarth::Runtime)
endif()
if(SHDISPLAY_ENABLE_VIDEO)
    target_compile_definitions(ShDisplayCore PUBLIC ZQXXXT_ENABLE_VIDEO)
    target_link_libraries(ShDisplayCore PUBLIC VideoSDK::Runtime)
endif()
qt_add_executable(ShDisplayViewer src/main.cpp resources/ZqxxxtClient.qrc)
target_link_libraries(ShDisplayViewer PRIVATE ShDisplayCore)
# The default project lives beside the executable. Rebuilds only add missing files.
add_custom_target(ShDisplayReleaseConfig
    COMMAND ${CMAKE_COMMAND}
        "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DOUTPUT=$<TARGET_FILE_DIR:ShDisplayViewer>"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Prepare-Release.cmake"
    VERBATIM)
add_dependencies(ShDisplayViewer ShDisplayReleaseConfig)
if(FREEEARTHSDK_PLUGIN_TARGETS_FOR_VIEWER)
    foreach(_plugin IN LISTS FREEEARTHSDK_PLUGIN_TARGETS_FOR_VIEWER)
        if(_plugin STREQUAL "FePluginDependent")
            set(_destination "$<TARGET_FILE_DIR:ShDisplayViewer>")
        else()
            set(_destination "$<TARGET_FILE_DIR:ShDisplayViewer>/${FREEEARTHSDK_PLUGIN_SUBDIR}")
        endif()
        add_custom_command(TARGET ShDisplayViewer POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E make_directory "${_destination}"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:${_plugin}>" "${_destination}"
            VERBATIM)
    endforeach()
endif()
if(WIN32)
    enable_language(RC)
    set(SHDISPLAY_ICON_ICO_FILE "${CMAKE_CURRENT_SOURCE_DIR}/resources/application/shdisplayviewer.ico")
    configure_file(resources/application/ShDisplayViewer.rc.in
        "${CMAKE_CURRENT_BINARY_DIR}/ShDisplayViewer.rc" @ONLY)
    target_sources(ShDisplayViewer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ShDisplayViewer.rc")
    set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/ShDisplayViewer.rc" APPEND
        PROPERTY OBJECT_DEPENDS "${SHDISPLAY_ICON_ICO_FILE}")
    set_target_properties(ShDisplayViewer PROPERTIES WIN32_EXECUTABLE TRUE)
    add_custom_command(TARGET ShDisplayViewer POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:ShDisplayViewer> $<TARGET_FILE_DIR:ShDisplayViewer>
        COMMAND_EXPAND_LISTS VERBATIM)
    if(FREEEARTHSDK_RUNTIME_DIR)
        add_custom_command(TARGET ShDisplayViewer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${FREEEARTHSDK_RUNTIME_DIR}" "$<TARGET_FILE_DIR:ShDisplayViewer>" VERBATIM)
    endif()
    if(SHDISPLAY_ENABLE_VIDEO)
        add_custom_command(TARGET ShDisplayViewer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${VIDEOSDK_FFMPEG_RUNTIME_DIR}" "$<TARGET_FILE_DIR:ShDisplayViewer>" VERBATIM)
    endif()
endif()
# Application-owned effects and UI resources; no map tiles or original client configuration.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Data/texture/sky")
foreach(_name moon_1024x512.jpg nebula.jpg)
    configure_file(resources/environment/${_name} "${CMAKE_CURRENT_BINARY_DIR}/Data/texture/sky/${_name}" COPYONLY)
endforeach()
file(COPY resources/water/ DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/Data/texture/river" PATTERN "._*" EXCLUDE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Data/model")
configure_file(resources/flight/wrj.ive "${CMAKE_CURRENT_BINARY_DIR}/Data/model/wrj.ive" COPYONLY)
file(COPY assets/ DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/Data" PATTERN "._*" EXCLUDE)
if(BUILD_TESTING)
    add_test(NAME ShDisplayViewer.ReleaseConfig COMMAND ${CMAKE_COMMAND}
        "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}" "-DBINARY=${CMAKE_CURRENT_BINARY_DIR}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/ReleaseConfigTests.cmake")
    foreach(_test DynamicPlans DeploymentXmlStore FlightRouteXmlStore MapTools NormalAnnotations LayerAddDialog EnvironmentSettingsDialog Standalone PlaybackIntegration MarkerActionRuntime)
        if(MSVC)
            set_property(SOURCE tests/${_test}Tests.cpp APPEND PROPERTY OBJECT_DEPENDS ${_client_headers})
        endif()
        qt_add_executable(ShDisplay${_test}Tests tests/${_test}Tests.cpp resources/ZqxxxtClient.qrc)
        target_link_libraries(ShDisplay${_test}Tests PRIVATE ShDisplayCore)
        add_test(NAME ShDisplayViewer.${_test} COMMAND ShDisplay${_test}Tests)
        set_tests_properties(ShDisplayViewer.${_test} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    endforeach()
    target_compile_definitions(ShDisplayStandaloneTests PRIVATE
        SHDISPLAY_DEFAULT_PROJECT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/resources/default-project")
endif()
install(TARGETS ShDisplayViewer RUNTIME DESTINATION bin)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Data" DESTINATION bin)
